-- Drag
local function makeDraggable(handle, target)
	local dragging, startInput, startPos

	track(handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1
			or i.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			startInput = i.Position
			startPos = target.Position
		end
	end))

	track(UIS.InputChanged:Connect(function(i)
		if dragging and (
			i.UserInputType == Enum.UserInputType.MouseMovement
			or i.UserInputType == Enum.UserInputType.Touch
		) then

			local d = i.Position - startInput

			target.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + d.X,
				startPos.Y.Scale,
				startPos.Y.Offset + d.Y
			)
		end
	end))

	track(UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1
			or i.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end))
end

local function setAimLine(head)
	if not S.espLine or not head then
		hideAimLine()
		return
	end

	local cam = Workspace.CurrentCamera
	if not cam then
		hideAimLine()
		return
	end

	local p, onScreen = cam:WorldToViewportPoint(head.Position)
	if not onScreen then
		hideAimLine()
		return
	end

	local center = cam.ViewportSize / 2
	local target = Vector2.new(p.X, p.Y)
	local delta = target - center
	local distance = delta.Magnitude

	if distance < 1 then
		center = center + Vector2.new(1, 0)
		delta = target - center
		distance = delta.Magnitude
	end

	aimLine.Position = UDim2.fromOffset(center.X, center.Y)
	aimLine.Size = UDim2.fromOffset(distance, 2)
	aimLine.Rotation = math.deg(math.atan2(delta.Y, delta.X))
	aimDot.Position = UDim2.fromOffset(target.X, target.Y)
	aimLine.Visible = true
	aimDot.Visible = true
end

		Size = UDim2.new(1, -20, 0, 6),

		Size = UDim2.fromScale((def - min) / (max - min), 1),
		BackgroundColor3 = C.accent

		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(1, 0.5),

		BackgroundColor3 = Color3.new(1, 1, 1)
	}, fill)

	corner(knob, 6)

	local dragging = false

	local function update(x)
		local rel = math.clamp(
			(x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X,
			0,
			1
		)

		local v = math.floor(
			min + (max - min) * rel + 0.5
		)

		fill.Size = UDim2.fromScale(rel, 1)
		lbl.Text = text .. ": " .. v
		cb(v)
	end

	bar.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1
			or i.UserInputType == Enum.UserInputType.Touch then

			dragging = true
			update(i.Position.X)
		end
	end)

	track(UIS.InputChanged:Connect(function(i)
		if dragging and (
			i.UserInputType == Enum.UserInputType.MouseMovement
			or i.UserInputType == Enum.UserInputType.Touch
		) then
			update(i.Position.X)
		end
	end))

	track(UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1
			or i.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end))

-- Helpers
local function getChar()
	local c = LP.Character
	if not c then return end

	return c,
		c:FindFirstChildOfClass("Humanoid"),
		c:FindFirstChild("HumanoidRootPart")
			or c:FindFirstChild("Torso")
end

local npcs = {}

local function tryAddNPC(hum)
	local m = hum.Parent

	if m and m:IsA("Model")
		and m ~= LP.Character
		and not Players:GetPlayerFromCharacter(m) then

		npcs[m] = true
	end
end

for _, d in ipairs(Workspace:GetDescendants()) do
	if d:IsA("Humanoid") then
		tryAddNPC(d)
	end
end

track(Workspace.DescendantAdded:Connect(function(d)
	if d:IsA("Humanoid") then
		task.defer(tryAddNPC, d)
	end
end))

local function alive(m)
	if not m or not m.Parent then return nil end

	local h = m:FindFirstChildOfClass("Humanoid")

	if not h or h.Health <= 0 then
		return nil
	end

	return h
end

local defaultFov = nil
local function setZoom(v)
	S.zoom = v
	local cam = Workspace.CurrentCamera
	if not cam then return end
	if v then
		if not defaultFov then defaultFov = cam.FieldOfView end
		cam.FieldOfView = math.clamp(S.zoomVal, 40, 120)
	else
		if defaultFov then cam.FieldOfView = defaultFov end
		defaultFov = nil
	end
end

-- ถอดจิต: ควบคุมกล้องแยกจากตัวละคร แล้ววาร์ปไปตำแหน่งกล้องได้
local spiritConn
local spiritPosValue
local function stopSpirit()
	if spiritConn then spiritConn:Disconnect(); spiritConn = nil end
	local cam = Workspace.CurrentCamera
	if cam then
		cam.CameraType = Enum.CameraType.Custom
		local _, hum = getChar()
		if hum then cam.CameraSubject = hum end
	end
	S.spirit = false
	S.spiritPos = nil
	spiritPad.Visible = false
	for k in pairs(spiritHeld) do spiritHeld[k] = false end
	if spiritPosValue then spiritPosValue.Text = "ตำแหน่งถอดจิต: ยังไม่ได้บันทึก" end
end

local function startSpirit()
	stopSpirit()
	local cam = Workspace.CurrentCamera
	if not cam then return end
	cam.CameraType = Enum.CameraType.Scriptable
	S.spirit = true
	S.spiritPos = cam.CFrame
	spiritPad.Visible = true
	spiritConn = RunService.RenderStepped:Connect(function(dt)
		if not S.spirit or not cam.Parent then return end
		local cf = cam.CFrame
		local move = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) or spiritHeld.f then move += cf.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.S) or spiritHeld.b then move -= cf.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.A) or spiritHeld.l then move -= cf.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.D) or spiritHeld.r then move += cf.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.Space) or spiritHeld.u then move += Vector3.yAxis end
		if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or spiritHeld.d then move -= Vector3.yAxis end
		if move.Magnitude > 0 then
			cam.CFrame = cf + move.Unit * S.spiritVal * dt
		end
		S.spiritPos = cam.CFrame
	end)
end

local function setSpirit(v)
	if v then startSpirit() else stopSpirit() end
end

local function warpToSpirit()
	if not S.spiritPos then notify("ยังไม่มีตำแหน่งถอดจิต") return end
	local _, _, root = getChar()
	if root then
		root.CFrame = CFrame.new(S.spiritPos.Position + Vector3.new(0, 3, 0))
		notify("วาร์ปไปยังตำแหน่งที่ถอดจิตไว้แล้ว")
	end
end

local function stopFly()
	if flyBV then
		flyBV:Destroy()
		flyBV = nil
	end

	if flyBG then
		flyBG:Destroy()
		flyBG = nil
	end

	local _, hum = getChar()

	if hum then
		hum.PlatformStand = false
	end

	flyPad.Visible = false
	upHeld, downHeld = false, false
end

local function startFly()
	stopFly()

	local _, hum, root = getChar()

	if not (hum and root) then return end

	flyBV = new("BodyVelocity", {
		MaxForce = Vector3.new(1e9, 1e9, 1e9),
		Velocity = Vector3.zero
	}, root)

	flyBG = new("BodyGyro", {
		MaxTorque = Vector3.new(1e9, 1e9, 1e9),
		P = 9e4,
		CFrame = root.CFrame
	}, root)

	hum.PlatformStand = true
	flyPad.Visible = true
end

local function setFly(v)
	S.fly = v

	if v then
		startFly()
	else
		stopFly()
	end
end

-- Aimlock
local function updateFovVisible()
	fovCircle.Visible = S.lockP or S.lockN
end

RunService:BindToRenderStep(
	"KTHubAim",
	Enum.RenderPriority.Camera.Value + 1,
	function()

		if not (S.lockP or S.lockN) then
			hideAimLine()
			return
		end

		local cam = Workspace.CurrentCamera

		if not cam then return end

		local center = cam.ViewportSize / 2
		local best, bestD

		local function consider(m)
			if not alive(m) then return end

			local head = m:FindFirstChild("Head")

			if not head then return end

			local p, onScreen =
				cam:WorldToViewportPoint(head.Position)

			if not onScreen then return end

			local d =
				(Vector2.new(p.X, p.Y) - center).Magnitude

			if d <= S.fov and
				(not bestD or d < bestD) then

				best, bestD = head, d
			end
		end

		if S.lockN then
			for m in pairs(npcs) do
				consider(m)
			end
		end

		if S.lockP then
			for _, p in ipairs(Players:GetPlayers()) do

				if p ~= LP and p.Character then

					local sameTeam =
						S.teamCheck
						and LP.Team ~= nil
						and p.Team == LP.Team

					if not sameTeam then
						consider(p.Character)
					end
				end
			end
		end

		if best then
			setAimLine(best)
			local goal =
				CFrame.lookAt(
					cam.CFrame.Position,
					best.Position
				)

			cam.CFrame =
				cam.CFrame:Lerp(goal, 0.5)
		else
			hideAimLine()
		end
	end
)

-- Invisible
local savedTransparency = {}

local function setInvisible(v)
	S.invisible = v

	local c = LP.Character
	if not c then return end

	if v then
		for _, obj in ipairs(c:GetDescendants()) do
			if obj:IsA("BasePart")
				or obj:IsA("Decal")
				or obj:IsA("Texture") then

				if savedTransparency[obj] == nil then
					savedTransparency[obj] = obj.Transparency
				end

				obj.Transparency = 1
			end
		end

		for obj, tr in pairs(savedTransparency) do
			if obj and obj.Parent then
				obj.Transparency = tr
			end

			savedTransparency[obj] = nil
		end
	end
end

-- Click teleport
local function clickTeleport(pos)
	local _, _, root = getChar()
	if not root then return end

	local cam = Workspace.CurrentCamera
	if not cam then return end

	local ray = cam:ViewportPointToRay(pos.X, pos.Y)

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {
		LP.Character
	}

	local hit = Workspace:Raycast(
		ray.Origin,
		ray.Direction * 1000,
		params
	)

	if hit then
		root.CFrame = CFrame.new(
			hit.Position + Vector3.new(0, 3, 0)
		)
	end
end

local function flingSelected()

		selectedPlr
		and selectedPlr.Character
	) then

		notify("ยังไม่ได้เลือกผู้เล่น")
		return
	end

	local targetRoot =
		selectedPlr.Character:
		FindFirstChild("HumanoidRootPart")

	local _, _, root = getChar()

	if not (targetRoot and root) then
		return
	end

	local original = root.CFrame

	root.CFrame =
		targetRoot.CFrame *
		CFrame.new(0, 0, -2)

	local bv = Instance.new("BodyVelocity")

	bv.MaxForce =
		Vector3.new(1e9, 1e9, 1e9)

	bv.Velocity =
		targetRoot.CFrame.LookVector * 180
		+ Vector3.new(0, 35, 0)

	bv.Parent = root

	task.wait(0.12)

	bv:Destroy()

	if root and root.Parent then
		root.CFrame = original
	end
end

	notify("กำลังลดแลค...")

	task.spawn(function()

			settings().Rendering.QualityLevel =
				Enum.QualityLevel.Level01
		end)

		for _, e in ipairs(Lighting:GetChildren()) do
			if e:IsA("PostEffect") then
				e.Enabled = false
			end
		end

		for _, d in ipairs(workspace:GetDescendants()) do
			if d:IsA("ParticleEmitter")
				or d:IsA("Trail") then

				d.Enabled = false

			elseif d:IsA("BasePart") then

				d.CastShadow = false

	end)
end

-- Server Hop
local function serverHop()
	notify("กำลังหาเซิร์ฟเวอร์...")

	local ok, res = pcall(function()
		return game:HttpGet(
			("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):
			format(game.PlaceId)
		)
	end)

	if not ok then
		notify("ดึงรายชื่อเซิร์ฟไม่สำเร็จ")
		return
	end

	local ok2, data =
		pcall(function()
			return HttpService:JSONDecode(res)
		end)

	if ok2 and data.data then
		for _, s in ipairs(data.data) do

			if s.id ~= game.JobId
				and s.playing < s.maxPlayers then

				TeleportService:
					TeleportToPlaceInstance(
						game.PlaceId,
						s.id,
						LP
					)

				return
			end
		end
	end

	notify("ไม่พบเซิร์ฟเวอร์")
end

-- Heartbeat
track(
	RunService.Heartbeat:Connect(function(dt)

		if dt > 0 then
			fpsValue =
				1 / dt
		end

		local _, hum, root = getChar()

		if S.speed and hum then
			hum.WalkSpeed = S.speedVal
		end

		if S.fly
			and flyBV
			and flyBG
			and root
			and hum then

			local cam = Workspace.CurrentCamera

			if cam then

				local look =
					cam.CFrame.LookVector

				local right =
					cam.CFrame.RightVector

				local flatLook =
					Vector3.new(
						look.X,
						0,
						look.Z
					).Unit

				local flatRight =
					Vector3.new(
						right.X,
						0,
						right.Z
					).Unit

				local md =
					hum.MoveDirection

				local vel =
					(
						look * md:Dot(flatLook)
						+
						right * md:Dot(flatRight)
					)
					* S.flyVal

				if upHeld then
					vel +=
						Vector3.new(
							0,
							S.flyVal,
							0
						)
				end

				if downHeld then
					vel -=
						Vector3.new(
							0,
							S.flyVal,
							0
						)
				end

				flyBV.Velocity = vel
				flyBG.CFrame = cam.CFrame
			end
		end

		if S.car and hum then

			local seat = hum.SeatPart

			if seat
				and seat:IsA("VehicleSeat")
				and seat.Throttle ~= 0 then

				local r =
					seat.AssemblyRootPart
					or seat

				r.AssemblyLinearVelocity +=
					seat.CFrame.LookVector
					* seat.Throttle
					* S.carVal
					* dt
			end
		end
	end)
)

-- Cleanup
local function cleanup()

	setSpeed(false)
	setFly(false)
	setZoom(false)
	stopSpirit()

	S.hitbox = false
	hideAimLine()

	setNoclip(false)
	setInvisible(false)
	setClickTeleport(false)
	setPickupOnce(false)

	pcall(function()
		RunService:UnbindFromRenderStep(
			"KTHubAim"
		)
	end)

	for _, c in ipairs(conns) do
		pcall(function()
			c:Disconnect()
		end)
	end

	if gui then
		gui:Destroy()
	end

	if overlay then
		overlay:Destroy()
	end
end

env.__KTHubCleanup = cleanup

closeBtn.MouseButton1Click:Connect(cleanup)

notify(
	"ขอโทษ HUB v3.4 โหลดเสร็จเรียบร้อย!"
)
